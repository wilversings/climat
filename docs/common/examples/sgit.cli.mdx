```clidsl title="sgit.cli"
sgit {
    sub acp(amend a: flag) {
        act sh {
            git add . &&
            git commit @{amend ? "--amend"} &&
            git push @{amend ? "--force"}
        }
    }
    sub cf(branch: arg, force f: flag) {
        act sh {
            git checkout feature/@{branch} @{force ? "--force"}
        }
    }
}
```
