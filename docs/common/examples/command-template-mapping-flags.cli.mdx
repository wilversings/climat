```clidsl title=mapping-flags.cli
hello-world(goodDay: flag) {
    sub foo {
        act sh { echo Hello World @{goodDay} }
    }
    sub bar {
        act sh { echo Hello World @{goodDay ? "today-is-a-good-day"} }
    }
    sub baz {
        act sh { echo Hello World @{!goodDay ? "today-is-NOT-a-good-day"} }
    }
}
```
